Core Statutory & Financial Capabilities

Every validator and formatter is built to execute with zero heap allocations on hot verification paths while preventing floating-point rounding drift and timing side-channels.

Exact INR Paise Math Money, NewMoney, Split, Allocate Stores currency as int64 paise (1 INR = 100 paise). Hare-Niemeyer proportional allocation guarantees sum of parts equals total with zero remainder loss.

Verified End-to-End Checkout & KYC Example

The complete runnable program in examples/checkout_verification/main.go exercises KYC verification, GST invoice splitting, UPI intent URI generation, Cashfree webhook authentication, and MCA Schedule III aging in a single pipeline: 

package main

import (
	"context"
	"fmt"
	"time"

	fintechin "github.com/abeta-dev/go-fintech-india"
)

func main() {
	// 1. Statutory KYC: GSTIN Mod-36, PAN Entity Classification & UIDAI Aadhaar Masking
	gstin := "29AACCG0527D1Z0"
	stateName, _ := fintechin.StateName(fintechin.StateCode(gstin))
	pan := fintechin.ExtractPAN(gstin)
	entityType, _ := fintechin.EntityType(pan)

	resolver := fintechin.NewOfflineIFSCResolver()
	branch, _ := resolver.Resolve(context.Background(), "HDFC0000123")

	fmt.Printf("KYC: GSTIN=%s (%s), PAN=%s (%s), Bank=%s\n",
		gstin, stateName, fintechin.MaskPAN(pan), entityType, branch.Bank)

	// 2. Exact Paise Invoice Math & Remainder-Safe CGST/SGST Split
	subtotal, _ := fintechin.ParseINR("₹1,25,000.00")
	gstTotal := fintechin.NewMoney(subtotal.Paise() * 18 / 100)
	cgstSgst, _ := gstTotal.Split(2)
	grandTotal := subtotal.Add(gstTotal)

	fmt.Printf("Total: %s (CGST: ₹%s, SGST: ₹%s)\n",
		fintechin.FormatINRSymbol(grandTotal.Paise()),
		fintechin.FormatINR(cgstSgst[0].Paise()),
		fintechin.FormatINR(cgstSgst[1].Paise()))
	fmt.Printf("Legal Words: %s\n", fintechin.InWords(grandTotal))

	// 3. NPCI UPI Intent URI Generation & MCA Schedule III Receivables Aging
	upiURI, _ := fintechin.GenerateUPIURI(fintechin.UPIParams{
		PayeeAddress: "abeta.vendor@okhdfcbank",
		PayeeName:    "Abeta Vendor Portal",
		Amount:       grandTotal.String(),
		RefID:        "INV-2026-001",
	})
	invDate := time.Date(2026, time.May, 20, 0, 0, 0, 0, time.UTC)
	fy := fintechin.FYFromDate(invDate)
	fmt.Printf("Settlement: %s | %s (%s)\n", upiURI, fy.Label(), fy.AssessmentYear())
}
